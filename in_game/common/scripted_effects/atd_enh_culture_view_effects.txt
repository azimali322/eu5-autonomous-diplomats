# ==========================================================================
# [ATD-ENH] Cultural Views. See atd_enh_cmm_effects.txt for the overview.
#
# Builds favors with countries that lead their culture (Curry Favors for
# Cultural Views), spends them on the game's Improve Cultural View diplomatic
# action the day they reach 50 (Auto Improve Cultural View), and surfaces
# what is left to do in alerts and the Cultural Views window
# (gui/atd_enh/atd_enh_culture_view_window.gui).
# Like every automated action in the mod, both spend diplomats only while you
# keep the Reserve settings (Settings tab): more than Minimum Diplomats Reserve
# banked - for Auto Improve, its own Improve Cultural View Reserve, which is
# never higher - and a monthly gain of at least Minimum Diplomat Gain.
#
# Monthly order: atd_enh_run_prioritized_pass (atd_enh_priority_effects.txt)
# runs atd_enh_cultural_view_spend first, before any other action claims a
# diplomat, because since 1.4.1 favors decay by 1% a month (FAVORS_DECAY): a
# curry stops at 50, and a month later the 50 is gone.
# atd_enh_cultural_view_build runs wherever "Curry Favors for Cultural Views"
# sits in the Automation Priority list, and the reminder is refreshed last, so
# it only lists what the automation left for you.
# Every effect here runs in the player country's scope (root).
# ==========================================================================

# ----- Auto Improve Cultural View -------------------------------------------
# Uses Improve Cultural View on every country that qualifies right now (see
# atd_enh_can_auto_improve_now: it leads its culture, would accept, you have the
# 50 favors it costs, and it was not tried in the last two months). Runs every
# day (atd_enh_cultural_view_daily) as well as first in the monthly pass:
# curries stop exactly at 50 and favors decay 1% a month, so the 50 can last
# less than a month - a monthly check missed it.
# It keeps the Reserve settings like every other action, with its own diplomat
# reserve (the player's calls on 2026-10-18): it needs more diplomats than
# Improve Cultural View Reserve - or Minimum Diplomats Reserve, if that is
# lower - and a monthly gain of at least Minimum Diplomat Gain. Its reserve can
# sit below the others' so that a curry reaching 50 is spent while the other
# automations are holding diplomats back (reported: Mali reached 50 with fewer
# diplomats than the reserve of 4, and the 50 decayed).
# Each attempt is checked straight away (atd_enh_improve_cultural_view_now). One
# that did not go through is not repeated for two months, and is checked again
# when that wait ends (atd_enh_verify_cultural_view_attempts), so the Mod Action
# Log always ends with whether the view really improved.
atd_enh_cultural_view_spend = {
	atd_enh_verify_cultural_view_attempts = yes
	if = {
		limit = {
			has_variable = atd_enh_auto_improve_cv_enabled
			is_dominant_country_of = culture
		}
		save_scope_as = atd_enh_actor
		# The lower of Improve Cultural View Reserve and Minimum Diplomats
		# Reserve, plus the diplomat the action uses. Until the setting is
		# registered, Minimum Diplomats Reserve alone.
		set_variable = { name = atd_enh_cv_threshold value = var:atd_min_diplomats }
		if = {
			limit = {
				has_variable = atd_enh_cv_min_diplomats
				var:atd_enh_cv_min_diplomats < var:atd_min_diplomats
			}
			set_variable = { name = atd_enh_cv_threshold value = var:atd_enh_cv_min_diplomats }
		}
		change_variable = { name = atd_enh_cv_threshold add = 1 }
		while = {
			count = 10
			limit = {
				num_of_diplomats >= var:atd_enh_cv_threshold
				atd_meets_min_diplomat_gain = yes
				any_country_in_diplomatic_range = { atd_enh_can_auto_improve_now = yes }
			}
			ordered_country_in_diplomatic_range = {
				limit = { atd_enh_can_auto_improve_now = yes }
				order_by = num_locations
				max = 1
				save_scope_as = atd_enh_cv_target
				scope:atd_enh_actor = { atd_enh_improve_cultural_view_now = yes }
			}
		}
		atd_enh_explain_cultural_view_waits = yes
	}
}

# Checks every attempt that had not gone through and whose two-month wait is
# over: did the target's culture's view of yours improve after all? Logs and
# announces it if so, or logs that it did not go through, then forgets the
# attempt. Scope: player country (root).
atd_enh_verify_cultural_view_attempts = {
	if = {
		limit = { has_variable_list = atd_enh_cv_pending }
		every_in_list = {
			variable = atd_enh_cv_pending
			limit = { NOT = { has_variable = atd_enh_cv_attempt } }
			root = { add_to_variable_list = { name = atd_enh_cv_done_tmp target = prev } }
		}
	}
	if = {
		limit = { has_variable_list = atd_enh_cv_done_tmp }
		every_in_list = {
			variable = atd_enh_cv_done_tmp
			save_scope_as = cmf_log_arg2
			if = {
				limit = {
					has_variable = atd_enh_cv_view_before
					atd_enh_cv_view_unchanged = yes
				}
				root = { cmf_log_with_scope_arg = { action = atd_enh_log_cv_failed } }
			}
			else_if = {
				limit = { has_variable = atd_enh_cv_view_before }
				if = {
					limit = { has_variable = atd_enh_curry_attempt }
					remove_variable = atd_enh_curry_attempt
				}
				root = { atd_enh_report_cultural_view_improved = yes }
			}
			if = {
				limit = { has_variable = atd_enh_cv_view_before }
				remove_variable = atd_enh_cv_view_before
			}
			root = { remove_list_variable = { name = atd_enh_cv_pending target = prev } }
		}
		clear_variable_list = atd_enh_cv_done_tmp
	}
}

# Records a cultural view Auto Improve raised: the Mod Action Log entry, and
# the "Cultural View Improved" alert if cultural view alerts are on.
# Scope: player country. Reads scope:cmf_log_arg2 (the country improved).
atd_enh_report_cultural_view_improved = {
	cmf_log_with_scope_arg = { action = atd_enh_log_cv_improved }
	# Named by the alert's tooltip.
	set_variable = { name = atd_enh_cv_improved_last value = scope:cmf_log_arg2 }
	if = {
		limit = { has_variable = atd_enh_culture_view_notify_enabled }
		cmf_show_alert = { alert = atd_enh_alert_culture_view_improved }
	}
}

# Records the level its culture's view of yours is at now, to compare after the
# attempt. Scope: the target country. Root: the player.
atd_enh_save_cv_view = {
	if = {
		limit = { culture = { cultural_view = { target = root.culture value = enemy } } }
		set_variable = { name = atd_enh_cv_view_before value = flag:enemy }
	}
	else_if = {
		limit = { culture = { cultural_view = { target = root.culture value = negative } } }
		set_variable = { name = atd_enh_cv_view_before value = flag:negative }
	}
	else_if = {
		limit = { culture = { cultural_view = { target = root.culture value = neutral } } }
		set_variable = { name = atd_enh_cv_view_before value = flag:neutral }
	}
	else_if = {
		limit = { culture = { cultural_view = { target = root.culture value = positive } } }
		set_variable = { name = atd_enh_cv_view_before value = flag:positive }
	}
	else = {
		set_variable = { name = atd_enh_cv_view_before value = flag:kindred }
	}
}

# Says in the Mod Action Log why Auto Improve skipped a country you already have
# 50 favors with - at most once a month per country - so a skipped improvement
# is never silent. Only countries the action could apply to are reported (they
# lead their culture, aren't your subject, aren't at war with you, and don't
# already see your culture as kindred). Runs right after the improvements, so
# countries just improved are below 50 again, and countries just tried are in
# their two-month wait (their attempt has its own log entry); neither is
# reported. Reads atd_enh_cv_threshold, set by atd_enh_cultural_view_spend.
# Scope: player country (root).
atd_enh_explain_cultural_view_waits = {
	every_country_in_diplomatic_range = {
		limit = {
			is_dominant_country_of = culture
			NOT = { is_subject_of = root }
			NOT = { is_at_war_with = root }
			NOT = {
				culture = {
					cultural_view = {
						target = root.culture
						value = kindred
					}
				}
			}
			prev_favors_with_this >= 50
			NOT = { has_variable = atd_enh_cv_attempt }
			NOT = { has_variable = atd_enh_cv_wait_logged }
		}
		save_scope_as = cmf_log_arg2
		if = {
			limit = {
				OR = {
					is_rival_of = root
					root = { is_rival_of = prev }
				}
			}
			root = { cmf_log_with_scope_arg = { action = atd_enh_log_cv_wait_rival } }
		}
		else_if = {
			limit = { NOT = { opinion = { target = root value > 50 } } }
			root = { cmf_log_with_scope_arg = { action = atd_enh_log_cv_wait_opinion } }
		}
		else_if = {
			limit = { root = { NOT = { num_of_diplomats >= var:atd_enh_cv_threshold } } }
			root = { cmf_log_with_scope_arg = { action = atd_enh_log_cv_wait_diplomat } }
		}
		else_if = {
			limit = { root = { atd_meets_min_diplomat_gain = no } }
			root = { cmf_log_with_scope_arg = { action = atd_enh_log_cv_wait_gain } }
		}
		set_variable = { name = atd_enh_cv_wait_logged value = yes months = 1 }
	}
	# Favors stuck just under 50: curries cannot go past their trust in you
	# (CURRY_FAVORS_LOW_TRUST), so with trust below 50 they stall at about the
	# trust value - which the game may display as "50" while the action still
	# needs a full 50.
	every_country_in_diplomatic_range = {
		limit = {
			is_dominant_country_of = culture
			NOT = { is_subject_of = root }
			NOT = {
				culture = {
					cultural_view = {
						target = root.culture
						value = kindred
					}
				}
			}
			prev_favors_with_this >= 45
			prev_favors_with_this < 50
			NOT = { trust = { target = root value >= 50 } }
			NOT = { has_variable = atd_enh_cv_wait_logged }
		}
		save_scope_as = cmf_log_arg2
		root = { cmf_log_with_scope_arg = { action = atd_enh_log_cv_wait_trust } }
		set_variable = { name = atd_enh_cv_wait_logged value = yes months = 1 }
	}
}

# ----- Daily check ----------------------------------------------------------
# Runs every day (scheduled by atd_enh_schedule_daily_checks), so a country is
# improved - or the reminder shows - the day its favors reach 50, not up to a
# month later. Settings > Enabled is the mod-wide master switch.
# Scope: player country (root).
atd_enh_cultural_view_daily = {
	if = {
		limit = { has_variable = atd_enabled }
		atd_enh_cultural_view_spend = yes
		# Refresh the reminder only when the number of countries ready to improve
		# changed, so a dismissed alert is not re-raised every day for nothing.
		if = {
			limit = { has_variable = atd_enh_culture_view_notify_enabled }
			set_variable = { name = atd_enh_cv_count_now value = 0 }
			if = {
				limit = { is_dominant_country_of = culture }
				every_country_in_diplomatic_range = {
					limit = { atd_enh_can_improve_cultural_view = yes }
					root = { change_variable = { name = atd_enh_cv_count_now add = 1 } }
				}
			}
			if = {
				limit = {
					OR = {
						NOT = { has_variable = atd_enh_cv_n }
						NOT = { var:atd_enh_cv_count_now = var:atd_enh_cv_n }
					}
				}
				atd_enh_update_culture_view_notify = yes
			}
			remove_variable = atd_enh_cv_count_now
		}
	}
}

# Performs Improve Cultural View on scope:atd_enh_cv_target, then checks right
# away whether its culture's view of yours really moved.
# mode = accept, as in every perform_diplomatic_action call in the game's own
# script (peace_treaties/hegemon_demands.txt) and in other mods that automate
# country interactions. Without a mode the action is only a proposal ("can be
# rejected": DIPLO_ACTION_IS_PROPOSAL), and from script it never went through:
# a 1360.1.1 save (reported 2026-10-18) logged 16 attempts on Kanem, Mali and
# Tunis, each costing a diplomat, with every favor still unspent and no view
# changed. Their consent still counts: only countries that would accept are
# picked (atd_enh_cultural_view_target: opinion of you above 50, no rivals).
# Every attempt starts a two-month wait with that country (atd_enh_cv_attempt),
# so even a failure can never be repeated daily again.
# Scope: player country (scope:atd_enh_actor).
atd_enh_improve_cultural_view_now = {
	scope:atd_enh_cv_target = {
		atd_enh_save_cv_view = yes
		set_variable = { name = atd_enh_cv_attempt value = yes months = 2 }
	}
	perform_diplomatic_action = {
		type = improve_cultural_view
		actor = scope:atd_enh_actor
		recipient = scope:atd_enh_cv_target
		mode = accept
	}
	scope:atd_enh_cv_target = {
		save_scope_as = cmf_log_arg2
		if = {
			limit = { atd_enh_cv_view_unchanged = no }
			remove_variable = atd_enh_cv_view_before
			# The curry that built these favors is over; let Curry Favors start the
			# next one right away instead of waiting out its 6-month safety marker.
			if = {
				limit = { has_variable = atd_enh_curry_attempt }
				remove_variable = atd_enh_curry_attempt
			}
			root = {
				atd_enh_report_cultural_view_improved = yes
				# The game's own cost should have taken the 50 favors. Say so if it
				# did not, so a free improvement shows up in the log (the two-month
				# wait already stops it from repeating).
				if = {
					limit = { scope:cmf_log_arg2 = { prev_favors_with_this >= 50 } }
					cmf_log_with_scope_arg = { action = atd_enh_log_cv_not_charged }
				}
			}
		}
		else = {
			# Not applied (yet): logged now, and checked again when the two months
			# are up (atd_enh_verify_cultural_view_attempts).
			root = {
				add_to_variable_list = { name = atd_enh_cv_pending target = prev }
				cmf_log_with_scope_arg = { action = atd_enh_log_cv_pending }
			}
		}
	}
}

# ----- Curry Favors for Cultural Views --------------------------------------
# Keeps up to "Countries to Curry at Once" curries running with countries that
# lead their culture (see atd_enh_can_start_curry), starting with cultures that
# are part of your own population, then the largest countries. The curry stops
# by itself at 50 favors (CURRY_FAVORS_MAX_FAVORS), which is exactly what Improve
# Cultural View costs.
# Running curries are counted from atd_enh_currying_seen, which two GUI
# trackers fill with every curry really in progress this month
# (atd_enh_curry_tracker.gui, which asks the game directly, and
# atd_enh_outliner_widgets.gui in the outliner) - so a curry that ended, was
# cancelled, or was started by hand is counted correctly. A curry this mod
# started in the last 6 months also counts while unconfirmed, so if both
# trackers ever failed (say, another mod broke them), it could still never run
# more curries than the setting allows.
# Curries are ongoing diplomat drains, so they respect both the diplomat reserve
# and the minimum monthly diplomat gain, like Improve Relations.
atd_enh_cultural_view_build = {
	if = {
		limit = {
			has_variable = atd_enh_curry_cv_enabled
			is_dominant_country_of = culture
		}
		save_scope_as = atd_enh_actor
		set_variable = { name = atd_enh_action_threshold value = { value = var:atd_min_diplomats add = 1 } }
		# Curries already running with countries that lead their culture...
		set_variable = { name = atd_enh_curry_n value = 0 }
		if = {
			limit = { has_variable_list = atd_enh_currying_seen }
			every_in_list = {
				variable = atd_enh_currying_seen
				limit = { is_dominant_country_of = culture }
				root = { change_variable = { name = atd_enh_curry_n add = 1 } }
			}
		}
		# ...plus ones this mod started in the last 6 months that no tracker
		# confirmed (the safety net described above).
		every_country_in_diplomatic_range = {
			limit = {
				has_variable = atd_enh_curry_attempt
				root = {
					OR = {
						NOT = { has_variable_list = atd_enh_currying_seen }
						NOT = {
							is_target_in_variable_list = {
								name = atd_enh_currying_seen
								target = prev
							}
						}
					}
				}
			}
			root = { change_variable = { name = atd_enh_curry_n add = 1 } }
		}
		while = {
			count = 10
			limit = {
				var:atd_enh_curry_n < var:atd_enh_curry_cv_count
				num_of_diplomats >= var:atd_enh_action_threshold
				atd_meets_min_diplomat_gain = yes
				any_country_in_diplomatic_range = { atd_enh_can_start_curry = yes }
			}
			# Cultures that are part of your own population first.
			if = {
				limit = {
					any_country_in_diplomatic_range = {
						atd_enh_can_start_curry = yes
						atd_enh_culture_is_in_our_country = yes
					}
				}
				ordered_country_in_diplomatic_range = {
					limit = {
						atd_enh_can_start_curry = yes
						atd_enh_culture_is_in_our_country = yes
					}
					order_by = num_locations
					max = 1
					save_scope_as = atd_enh_curry_target
					scope:atd_enh_actor = { atd_enh_start_curry = yes }
				}
			}
			else = {
				ordered_country_in_diplomatic_range = {
					limit = { atd_enh_can_start_curry = yes }
					order_by = num_locations
					max = 1
					save_scope_as = atd_enh_curry_target
					scope:atd_enh_actor = { atd_enh_start_curry = yes }
				}
			}
		}
		remove_variable = atd_enh_curry_n
	}
}

# Starts Curry Favors with scope:atd_enh_curry_target and records it.
# Scope: player country (scope:atd_enh_actor).
atd_enh_start_curry = {
	perform_diplomatic_action = {
		type = curry_favors
		actor = scope:atd_enh_actor
		recipient = scope:atd_enh_curry_target
	}
	# Counts toward the limit straight away and is not picked again this pass;
	# the outliner reports it from next month on.
	add_to_variable_list = { name = atd_enh_currying_seen target = scope:atd_enh_curry_target }
	change_variable = { name = atd_enh_curry_n add = 1 }
	scope:atd_enh_curry_target = {
		set_variable = { name = atd_enh_curry_attempt value = yes months = 6 }
		save_scope_as = cmf_log_arg2
	}
	cmf_log_with_scope_arg = { action = atd_enh_log_curry_started }
}

# Rebuilds atd_enh_culture_view_list: every country you can use Improve
# Cultural View on right now. The window binds its list straight to it.
# Rebuilt monthly and whenever the window opens, so it is current on screen.
atd_enh_rebuild_culture_view_list = {
	if = {
		limit = { has_variable_list = atd_enh_culture_view_list }
		clear_variable_list = atd_enh_culture_view_list
	}
	# The action is only available while you lead your own culture (its
	# potential block), so there is nothing to list otherwise.
	if = {
		limit = { is_dominant_country_of = culture }
		every_country_in_diplomatic_range = {
			limit = { atd_enh_can_improve_cultural_view = yes }
			root = {
				add_to_variable_list = {
					name = atd_enh_culture_view_list
					target = prev
				}
			}
		}
	}
}

# Opens or closes the Cultural Views window (it is shown while
# atd_enh_culture_view_window_open is set).
atd_enh_toggle_culture_view_window = {
	if = {
		limit = { has_variable = atd_enh_culture_view_window_open }
		remove_variable = atd_enh_culture_view_window_open
	}
	else = {
		atd_enh_rebuild_culture_view_list = yes
		set_variable = { name = atd_enh_culture_view_window_open value = 1 }
	}
}

# Refreshes the list, then raises the alert variant matching how many
# countries qualify, so its tooltip can name up to three of them. Only raises
# an alert while Settings > Enabled (the mod-wide master switch) and the
# reminder setting are both on.
atd_enh_update_culture_view_notify = {
	atd_enh_rebuild_culture_view_list = yes

	remove_variable = atd_enh_cv_t1
	remove_variable = atd_enh_cv_t2
	remove_variable = atd_enh_cv_t3
	set_variable = { name = atd_enh_cv_n value = 0 }

	if = {
		limit = {
			has_variable = atd_enabled
			has_variable = atd_enh_culture_view_notify_enabled
			has_variable_list = atd_enh_culture_view_list
		}
		every_in_list = {
			variable = atd_enh_culture_view_list
			root = {
				change_variable = { name = atd_enh_cv_n add = 1 }
				if = {
					limit = { NOT = { has_variable = atd_enh_cv_t1 } }
					set_variable = { name = atd_enh_cv_t1 value = prev }
				}
				else_if = {
					limit = { NOT = { has_variable = atd_enh_cv_t2 } }
					set_variable = { name = atd_enh_cv_t2 value = prev }
				}
				else_if = {
					limit = { NOT = { has_variable = atd_enh_cv_t3 } }
					set_variable = { name = atd_enh_cv_t3 value = prev }
				}
			}
		}
	}

	# Show only the variant that matches the count, and clear the others so a
	# stale one cannot sit on the bar naming countries that no longer qualify.
	if = {
		limit = { var:atd_enh_cv_n >= 4 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_1 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_2 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_3 }
		cmf_show_alert = { alert = atd_enh_alert_culture_view_more }
	}
	else_if = {
		limit = { var:atd_enh_cv_n = 3 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_1 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_2 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_more }
		cmf_show_alert = { alert = atd_enh_alert_culture_view_3 }
	}
	else_if = {
		limit = { var:atd_enh_cv_n = 2 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_1 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_3 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_more }
		cmf_show_alert = { alert = atd_enh_alert_culture_view_2 }
	}
	else_if = {
		limit = { var:atd_enh_cv_n = 1 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_2 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_3 }
		cmf_remove_alert = { alert = atd_enh_alert_culture_view_more }
		cmf_show_alert = { alert = atd_enh_alert_culture_view_1 }
	}
	else = {
		atd_enh_clear_culture_view_alerts = yes
	}
}

# Removes every Cultural View alert.
atd_enh_clear_culture_view_alerts = {
	cmf_remove_alert = { alert = atd_enh_alert_culture_view_1 }
	cmf_remove_alert = { alert = atd_enh_alert_culture_view_2 }
	cmf_remove_alert = { alert = atd_enh_alert_culture_view_3 }
	cmf_remove_alert = { alert = atd_enh_alert_culture_view_more }
}
