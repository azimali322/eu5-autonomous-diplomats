# ==========================================================================
# [ATD-ENH] Favor-economy engine: build favors by category, spend on culture.
# External contribution (azimali322). See atd_enh_cmm_effects.txt for the full
# list of base-file touch points and removal instructions.
#
# Runs from the base mod's monthly pass via atd_enh_run_prioritized_pass, so
# the Automation Priority order applies. Gated on Settings > Enabled (mod-wide
# master) and the Favors tab's "Improve Favors" toggle.
#
# Flow per tick:
#   1. Iterate the target-category priority list (atd__enh_favor_targets) in
#      the user's chosen order. For each category with count > 0, start
#      Curry Favors (the hardcoded curry_favors action: ongoing, ties up a
#      diplomat, +0.75 favors/month, needs 10+ relations, caps at 25 favors).
#   2. If Culture Improve is enabled and we lead our own culture, spend 50
#      favors via improve_cultural_view on targets that already owe us 50.
#
# GIFTS ARE NOT AUTOMATED: perform_diplomatic_action rejects send_gift
# ("Cannot find send_gift in country_interactions database or list of hardcoded
# diplo actions"), since it is built around its own amount-slider window. The
# mod instead names giftable countries in an alert for the player to act on.
#
# Diplomat budget: every action is gated on num_of_diplomats >= the base mod's
# reserve (atd_min_diplomats + 1) and its minimum monthly gain, sharing the
# same reserve semantics as the base categories.
#
# Every performed action is recorded to the CMF Mod Action Log (cmf_log_*), so
# what the automation did each month is visible in the log window.
# ==========================================================================

# Runs the three automations (Subjects / Favors / Improve Relations) in the
# user's Automation Priority order (Settings tab). Called from the base mod's
# atd_gui_sync_step via a marked [ATD-ENH] edit, replacing its fixed
# enforce-then-improve order. Each automation keeps its own enable gates.
# Scope: player country.
atd_enh_run_prioritized_pass = {
	cmm_for_each_list_item = {
		setting = atd__enh_priority
		effect = atd_enh_priority_each_item
	}
}

atd_enh_priority_each_item = {
	switch = {
		trigger = scope:cmm_list_current_item_value
		flag:subjects_pass = { atd_enforce_on_subjects = yes }
		flag:favors_pass = { atd_enh_favor_economy = yes }
		flag:improve_pass = {
			if = {
				limit = { has_variable = atd_improve_relations_enabled }
				atd_improve_relations = yes
			}
		}
	}

	cmf_suppress = { v = $i$ }
	cmf_suppress = { v = $setting$ }
}

# Entry point. Scope: player country.
atd_enh_favor_economy = {
	if = {
		limit = {
			# Settings > Enabled is the mod-wide master switch: nothing this mod
			# adds may act while it is off.
			has_variable = atd_enabled
			has_variable = atd_enh_favors_enabled
		}
		save_scope_as = actor_scope

		# Respect the base mod's diplomat reserve: threshold = reserve + 1.
		set_variable = { name = atd_enh_action_threshold value = { value = var:atd_min_diplomats add = 1 } }

		if = {
			limit = { has_variable_list = atd_enh_favor_done_this_tick }
			clear_variable_list = atd_enh_favor_done_this_tick
		}

		# Release curries that are no longer running so their category slot frees up.
		# atd_enh_currying_with is what makes a category's count behave as a cap on
		# concurrent diplomats, so a stale entry silently consumes a slot forever -
		# the category looks full while only a fraction of it is really currying.
		# An entry is dropped when the target reached the favor cap (done), the
		# country is gone, we went to war with it, or its opinion fell under the
		# game's CURRY_FAVORS_RELATION_REQUIRED floor of 10 - the last two cancel a
		# running curry engine-side, and nothing else would ever clear them.
		if = {
			limit = { has_variable_list = atd_enh_currying_with }
			every_in_list = {
				variable = atd_enh_currying_with
				limit = {
					OR = {
						NOT = { exists = this }
						AND = {
							exists = this
							OR = {
								NOT = { prev_favors_with_this < prev.var:atd_enh_curry_favor_cap }
								is_at_war_with = prev
								NOT = { opinion = { target = prev value >= 10 } }
							}
						}
					}
				}
				scope:actor_scope = {
					add_to_variable_list = { name = atd_enh_curry_stale_tmp target = prev }
				}
			}
			if = {
				limit = { has_variable_list = atd_enh_curry_stale_tmp }
				every_in_list = {
					variable = atd_enh_curry_stale_tmp
					scope:actor_scope = {
						remove_list_variable = { name = atd_enh_currying_with target = prev }
						remove_from_variable_map = { name = atd_enh_curry_type key = prev }
					}
				}
				clear_variable_list = atd_enh_curry_stale_tmp
			}
		}

		# Build favors per category, in priority order.
		cmm_for_each_list_item = {
			setting = atd__enh_favor_targets
			effect = atd_enh_favor_each_item
		}

		# Spend favors on cultural improvement.
		if = {
			limit = {
				has_variable = atd_enh_culture_improve_enabled
				# improve_cultural_view requires we lead our own culture.
				is_dominant_country_of = culture
			}
			atd_enh_culture_improve = yes
		}

		if = {
			limit = { has_variable_list = atd_enh_favor_done_this_tick }
			clear_variable_list = atd_enh_favor_done_this_tick
		}
		remove_variable = atd_enh_action_threshold
	}
}

# Per-list-item dispatch. Scope: player country (actor_scope saved).
atd_enh_favor_each_item = {
	switch = {
		trigger = scope:cmm_list_current_item_value
		flag:culture_leaders = { atd_enh_favor_culture_leaders = yes }
		flag:neighbors = { atd_enh_favor_neighbors = yes }
		flag:subjects = { atd_enh_favor_subjects = yes }
		flag:outraged = { atd_enh_favor_outraged = yes }
		flag:allies = { atd_enh_favor_allies = yes }
		flag:threatening = { atd_enh_favor_threatening = yes }
		flag:market_owners = { atd_enh_favor_market_owners = yes }
		flag:kingdoms = { atd_enh_favor_kingdoms = yes }
		flag:largest = { atd_enh_favor_largest = yes }
	}

	cmf_suppress = { v = $i$ }
	cmf_suppress = { v = $setting$ }
}

# Starts Curry Favors against scope:target_country and records it to the Mod
# Action Log.
# Scope: player country (actor_scope). Increments atd_enh_cat_done on success.
atd_enh_try_build_favor = {
	if = {
		limit = {
			has_variable = atd_enh_curry_enabled
			scope:target_country = { atd_enh_can_curry_target = yes }
		}
		perform_diplomatic_action = {
			type = curry_favors
			actor = scope:actor_scope
			recipient = scope:target_country
		}
		scope:target_country = { save_scope_as = cmf_log_arg2 }
		cmf_log_with_scope_arg = { action = atd_enh_log_curry }
		# Ongoing action: remember it so we do not re-issue it every month while
		# it runs, and remember WHICH category claimed it so that category's
		# limit counts it for as long as it runs. Both cleared once the target
		# reaches the favor cap.
		add_to_variable_list = {
			name = atd_enh_currying_with
			target = scope:target_country
		}
		add_to_variable_map = {
			name = atd_enh_curry_type
			key = scope:target_country
			value = flag:$type$
		}
		add_to_variable_list = {
			name = atd_enh_favor_done_this_tick
			target = scope:target_country
		}
		change_variable = { name = atd_enh_cat_done add = 1 }
	}

	# Suppress "used but never set" for the category flags passed via $type$.
	if = {
		limit = { always = no }
		cmf_suppress = { v = culture_leaders }
		cmf_suppress = { v = neighbors }
		cmf_suppress = { v = subjects }
		cmf_suppress = { v = outraged }
		cmf_suppress = { v = allies }
		cmf_suppress = { v = threatening }
		cmf_suppress = { v = market_owners }
		cmf_suppress = { v = kingdoms }
		cmf_suppress = { v = largest }
	}
}

# Seeds atd_enh_cat_done with the number of curries this category ALREADY has
# running, so the category slider behaves like the base mod's Diplomats column:
# a cap on concurrent diplomats tied up, not a number of new starts per month.
# Without this the engine started a fresh batch every month and the count grew
# without bound as diplomats freed up.
# Scope: player country. Args: type (category flag).
atd_enh_count_active_curries = {
	set_variable = { name = atd_enh_cat_done value = 0 }
	if = {
		limit = { has_variable_map = atd_enh_curry_type }
		every_key_in_variable_map = {
			variable = atd_enh_curry_type
			save_scope_as = atd_enh_ct_key
			scope:actor_scope = {
				if = {
					limit = { "variable_map(atd_enh_curry_type|scope:atd_enh_ct_key)" = flag:$type$ }
					change_variable = { name = atd_enh_cat_done add = 1 }
				}
			}
		}
	}
}

# Diagnostic: when a category ends a tick with fewer curries running than its
# configured count, record WHY in the Mod Action Log. From in-game the two causes
# look identical - the slider just fails to fill - so this separates "the
# diplomat budget stopped us" from "no country passed the eligibility filters"
# (dominant-of-culture / discovered / in diplomatic range / opinion >= 10 /
# owing less than the resume threshold).
# Scope: player country. Args: countvar (the category's count variable).
atd_enh_report_category_short = {
	if = {
		limit = { var:atd_enh_cat_done < var:$countvar$ }
		if = {
			limit = {
				OR = {
					NOT = { num_of_diplomats >= var:atd_enh_action_threshold }
					NOT = { modifier:monthly_diplomats >= var:atd_min_diplomat_gain }
				}
			}
			cmf_log = { action = atd_enh_log_short_diplomats }
		}
		else = {
			cmf_log = { action = atd_enh_log_short_no_target }
		}
	}
}

# ----- Per-category builders ----------------------------------------------
# Each: loop up to the category's count while the diplomat budget holds,
# picking one matching candidate per iteration (mirrors the base mod's
# category pickers).

atd_enh_favor_culture_leaders = {
	atd_enh_count_active_curries = { type = culture_leaders }
	while = {
		count = 10
		limit = {
			var:atd_enh_cat_done < var:atd_enh_ft_culture_leaders_count
			num_of_diplomats >= var:atd_enh_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
		}
		ordered_country_in_diplomatic_range = {
			limit = {
				is_dominant_country_of = culture
				atd_enh_is_favor_candidate = yes
			}
			order_by = { value = 0 }
			max = 1
			save_scope_as = target_country
			scope:actor_scope = { atd_enh_try_build_favor = { type = culture_leaders } }
		}
	}
	atd_enh_report_category_short = { countvar = atd_enh_ft_culture_leaders_count }
	remove_variable = atd_enh_cat_done
}

# ----- Culture Improve ----------------------------------------------------
# improve_cultural_view (costs 50 favors) on culture leaders not already at
# kindred, up to its own monthly count. The engine enforces the favor cost.
# Scope: player country (actor_scope saved).
atd_enh_culture_improve = {
	set_variable = { name = atd_enh_ci_done value = 0 }
	if = {
		limit = { has_variable_list = atd_enh_culture_improved_this_tick }
		clear_variable_list = atd_enh_culture_improved_this_tick
	}
	while = {
		count = 10
		limit = {
			var:atd_enh_ci_done < var:atd_enh_culture_improve_count
			num_of_diplomats >= var:atd_enh_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
			any_country_in_diplomatic_range = {
				atd_enh_is_culture_target = yes
			}
		}
		ordered_country_in_diplomatic_range = {
			limit = { atd_enh_is_culture_target = yes }
			order_by = { value = 0 }
			max = 1
			save_scope_as = target_country
			scope:actor_scope = {
				perform_diplomatic_action = {
					type = improve_cultural_view
					actor = scope:actor_scope
					recipient = scope:target_country
				}
				scope:target_country = { save_scope_as = cmf_log_arg2 }
				cmf_log_with_scope_arg = { action = atd_enh_log_culture }
				add_to_variable_list = {
					name = atd_enh_culture_improved_this_tick
					target = scope:target_country
				}
				change_variable = { name = atd_enh_ci_done add = 1 }
			}
		}
	}
	remove_variable = atd_enh_ci_done
	if = {
		limit = { has_variable_list = atd_enh_culture_improved_this_tick }
		clear_variable_list = atd_enh_culture_improved_this_tick
	}
}

# ----- Gift-availability notify alert -------------------------------------
# Rebuilds atd_enh_giftable_list: every country the player can send a gift to
# right now. The Giftable Countries window binds its scrolling list straight to
# this (datamodel), rendering one clickable row per country. Rebuilt on the
# monthly pulse and whenever the window is opened, so it is current on screen.
# Scope: player country (root).
atd_enh_rebuild_giftable_list = {
	if = {
		limit = { has_variable_list = atd_enh_giftable_list }
		clear_variable_list = atd_enh_giftable_list
	}
	every_country_in_diplomatic_range = {
		limit = { atd_enh_is_giftable_now = yes }
		root = {
			add_to_variable_list = {
				name = atd_enh_giftable_list
				target = prev
			}
		}
	}
}

# Opens/closes the Giftable Countries window (bound to atd_enh_gifts_window_open).
# Scope: player country.
atd_enh_toggle_gifts_window = {
	if = {
		limit = { has_variable = atd_enh_gifts_window_open }
		remove_variable = atd_enh_gifts_window_open
	}
	else = {
		atd_enh_rebuild_giftable_list = yes
		set_variable = { name = atd_enh_gifts_window_open value = 1 }
	}
}

# Names the dominant countries of a primary culture the player can gift right
# now (off the 10-year gift cooldown, still owing less than the gift cap) and
# raises the alert variant matching how many were found, so the tooltip can list
# them by name. Hooked to the monthly pulse, independent of the automation
# toggles. Scope: player country (root).
atd_enh_update_gift_notify = {
	# Rebuild the full clickable list used by the Giftable Countries window, plus
	# the first three name slots the alert tooltip reads.
	atd_enh_rebuild_giftable_list = yes

	remove_variable = atd_enh_gift_t1
	remove_variable = atd_enh_gift_t2
	remove_variable = atd_enh_gift_t3
	set_variable = { name = atd_enh_gift_n value = 0 }

	if = {
		limit = {
			# Settings > Enabled is the mod-wide master switch.
			has_variable = atd_enabled
			has_variable = atd_enh_gift_notify_enabled
			gold >= 25
		}
		every_country_in_diplomatic_range = {
			limit = { atd_enh_is_giftable_now = yes }
			root = {
				change_variable = { name = atd_enh_gift_n add = 1 }
				if = {
					limit = { NOT = { has_variable = atd_enh_gift_t1 } }
					set_variable = { name = atd_enh_gift_t1 value = prev }
				}
				else_if = {
					limit = { NOT = { has_variable = atd_enh_gift_t2 } }
					set_variable = { name = atd_enh_gift_t2 value = prev }
				}
				else_if = {
					limit = { NOT = { has_variable = atd_enh_gift_t3 } }
					set_variable = { name = atd_enh_gift_t3 value = prev }
				}
			}
		}
	}

	# Pick the alert variant that matches how many were found, and clear the
	# others so a stale one cannot sit on the bar reading empty slots.
	if = {
		limit = { var:atd_enh_gift_n >= 4 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
		cmf_show_alert = { alert = atd_enh_alert_giftable_more }
	}
	else_if = {
		limit = { var:atd_enh_gift_n = 3 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
		cmf_show_alert = { alert = atd_enh_alert_giftable_3 }
	}
	else_if = {
		limit = { var:atd_enh_gift_n = 2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
		cmf_show_alert = { alert = atd_enh_alert_giftable_2 }
	}
	else_if = {
		limit = { var:atd_enh_gift_n = 1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
		cmf_show_alert = { alert = atd_enh_alert_giftable_1 }
	}
	else = {
		cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
	}
}
