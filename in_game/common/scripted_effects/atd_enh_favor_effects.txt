# ==========================================================================
# [ATD-ENH] Favor-economy engine: build favors by category, spend on culture.
# External contribution (azimali322). See atd_enh_cmm_effects.txt for the full
# list of base-file touch points and removal instructions.
#
# Runs once per monthly tick from its own hook (atd_enh_on_actions.txt),
# independent of the base mod's improve-relations pass. Gated on the Favors
# tab's "Improve Favors" master toggle.
#
# Flow per tick:
#   1. Iterate the target-category priority list (atd__enh_favor_targets) in
#      the user's chosen order. For each category with count > 0, pick matching
#      countries and build favors with them, preferring Curry Favors
#      (influence_nation, +5 favors, non-great-powers, up to the curry cap)
#      and falling back to Send Gifts (send_gift, engine-decided amount, up to
#      the gift cap and the treasury spend cap).
#   2. If Culture Improve is enabled and we lead our own culture, spend favors
#      (improve_cultural_view, 50 favors each) on culture leaders.
#
# Diplomat budget: every action is gated on num_of_diplomats >= the base mod's
# reserve (atd_min_diplomats + 1) and its minimum monthly gain, sharing the
# same reserve semantics as the base categories.
#
# Every performed action is recorded to the CMF Mod Action Log (cmf_log_*), so
# what the automation did each month is visible in the log window.
# ==========================================================================

# Runs the three automations (Subjects / Favors / Improve Relations) in the
# user's Automation Priority order (Settings tab). Called from the base mod's
# atd_gui_sync_step via a marked [ATD-ENH] edit, replacing its fixed
# enforce-then-improve order. Each automation keeps its own enable gates.
# Scope: player country.
atd_enh_run_prioritized_pass = {
	cmm_for_each_list_item = {
		setting = atd__enh_priority
		effect = atd_enh_priority_each_item
	}
}

atd_enh_priority_each_item = {
	switch = {
		trigger = scope:cmm_list_current_item_value
		flag:subjects_pass = { atd_enforce_on_subjects = yes }
		flag:favors_pass = { atd_enh_favor_economy = yes }
		flag:improve_pass = {
			if = {
				limit = { has_variable = atd_improve_relations_enabled }
				atd_improve_relations = yes
			}
		}
	}

	cmf_suppress = { v = $i$ }
	cmf_suppress = { v = $setting$ }
}

# Entry point. Scope: player country.
atd_enh_favor_economy = {
	if = {
		limit = { has_variable = atd_enh_favors_enabled }
		save_scope_as = actor_scope

		# Respect the base mod's diplomat reserve: threshold = reserve + 1.
		set_variable = { name = atd_enh_action_threshold value = { value = var:atd_min_diplomats add = 1 } }

		if = {
			limit = { has_variable_list = atd_enh_favor_done_this_tick }
			clear_variable_list = atd_enh_favor_done_this_tick
		}

		# Build favors per category, in priority order.
		cmm_for_each_list_item = {
			setting = atd__enh_favor_targets
			effect = atd_enh_favor_each_item
		}

		# Spend favors on cultural improvement.
		if = {
			limit = {
				has_variable = atd_enh_culture_improve_enabled
				# improve_cultural_view requires we lead our own culture.
				is_dominant_country_of = culture
			}
			atd_enh_culture_improve = yes
		}

		if = {
			limit = { has_variable_list = atd_enh_favor_done_this_tick }
			clear_variable_list = atd_enh_favor_done_this_tick
		}
		remove_variable = atd_enh_action_threshold
	}
}

# Per-list-item dispatch. Scope: player country (actor_scope saved).
atd_enh_favor_each_item = {
	switch = {
		trigger = scope:cmm_list_current_item_value
		flag:culture_leaders = { atd_enh_favor_culture_leaders = yes }
		flag:neighbors = { atd_enh_favor_neighbors = yes }
		flag:subjects = { atd_enh_favor_subjects = yes }
		flag:outraged = { atd_enh_favor_outraged = yes }
		flag:allies = { atd_enh_favor_allies = yes }
		flag:threatening = { atd_enh_favor_threatening = yes }
		flag:market_owners = { atd_enh_favor_market_owners = yes }
		flag:kingdoms = { atd_enh_favor_kingdoms = yes }
		flag:largest = { atd_enh_favor_largest = yes }
	}

	cmf_suppress = { v = $i$ }
	cmf_suppress = { v = $setting$ }
}

# Attempts one favor-building action against scope:target_country, preferring
# curry over gift, and records it to the Mod Action Log.
# Scope: player country (actor_scope). Increments atd_enh_cat_done on success.
atd_enh_try_build_favor = {
	if = {
		limit = {
			has_variable = atd_enh_curry_enabled
			scope:target_country = { atd_enh_can_curry_target = yes }
		}
		perform_diplomatic_action = {
			type = influence_nation
			actor = scope:actor_scope
			recipient = scope:target_country
		}
		scope:target_country = { save_scope_as = cmf_log_arg2 }
		cmf_log_with_scope_arg = { action = atd_enh_log_curry }
		add_to_variable_list = {
			name = atd_enh_favor_done_this_tick
			target = scope:target_country
		}
		change_variable = { name = atd_enh_cat_done add = 1 }
	}
	else_if = {
		limit = {
			has_variable = atd_enh_gifts_enabled
			scope:target_country = { atd_enh_can_gift_target = yes }
		}
		# Logging is done from the game's own on_gift_sent hook rather than
		# optimistically here, so a log entry proves a gift actually happened.
		# The marker tells the hook this gift was ours (not a manual one); the
		# hook fires synchronously inside perform_diplomatic_action, so clearing
		# it straight after also cleans up when the action silently did nothing.
		set_variable = { name = atd_enh_gift_pending value = 1 }
		perform_diplomatic_action = {
			type = send_gift
			actor = scope:actor_scope
			recipient = scope:target_country
		}
		if = {
			limit = { has_variable = atd_enh_gift_pending }
			remove_variable = atd_enh_gift_pending
		}
		add_to_variable_list = {
			name = atd_enh_favor_done_this_tick
			target = scope:target_country
		}
		change_variable = { name = atd_enh_cat_done add = 1 }
	}
}

# Called from the game's on_gift_sent hook (root = sender, scope:recipient =
# target). Only logs gifts this mod initiated - a manual gift leaves
# atd_enh_gift_pending unset and is ignored.
atd_enh_on_gift_sent_log = {
	if = {
		limit = { has_variable = atd_enh_gift_pending }
		remove_variable = atd_enh_gift_pending
		# scope:target is the gift's GOLD AMOUNT here (not a country). The engine
		# picks the amount for script-performed gifts - there is no way to set the
		# slider from script - so record what it actually chose. Readable in the
		# console with: variable atd_enh_last_gift_amount
		set_variable = { name = atd_enh_last_gift_amount value = scope:target }
		scope:recipient = { save_scope_as = cmf_log_arg2 }
		cmf_log_with_scope_arg = { action = atd_enh_log_gift }
	}
}

# ----- Per-category builders ----------------------------------------------
# Each: loop up to the category's count while the diplomat budget holds,
# picking one matching candidate per iteration (mirrors the base mod's
# category pickers).

atd_enh_favor_culture_leaders = {
	set_variable = { name = atd_enh_cat_done value = 0 }
	while = {
		count = 10
		limit = {
			var:atd_enh_cat_done < var:atd_enh_ft_culture_leaders_count
			num_of_diplomats >= var:atd_enh_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
		}
		ordered_country_in_diplomatic_range = {
			limit = {
				is_dominant_country_of = culture
				atd_enh_is_favor_candidate = yes
			}
			order_by = { value = 0 }
			max = 1
			save_scope_as = target_country
			scope:actor_scope = { atd_enh_try_build_favor = yes }
		}
	}
	remove_variable = atd_enh_cat_done
}

# ----- Culture Improve ----------------------------------------------------
# improve_cultural_view (costs 50 favors) on culture leaders not already at
# kindred, up to its own monthly count. The engine enforces the favor cost.
# Scope: player country (actor_scope saved).
atd_enh_culture_improve = {
	set_variable = { name = atd_enh_ci_done value = 0 }
	if = {
		limit = { has_variable_list = atd_enh_culture_improved_this_tick }
		clear_variable_list = atd_enh_culture_improved_this_tick
	}
	while = {
		count = 10
		limit = {
			var:atd_enh_ci_done < var:atd_enh_culture_improve_count
			num_of_diplomats >= var:atd_enh_action_threshold
			modifier:monthly_diplomats >= var:atd_min_diplomat_gain
			any_country_in_diplomatic_range = {
				atd_enh_is_culture_target = yes
			}
		}
		ordered_country_in_diplomatic_range = {
			limit = { atd_enh_is_culture_target = yes }
			order_by = { value = 0 }
			max = 1
			save_scope_as = target_country
			scope:actor_scope = {
				perform_diplomatic_action = {
					type = improve_cultural_view
					actor = scope:actor_scope
					recipient = scope:target_country
				}
				scope:target_country = { save_scope_as = cmf_log_arg2 }
				cmf_log_with_scope_arg = { action = atd_enh_log_culture }
				add_to_variable_list = {
					name = atd_enh_culture_improved_this_tick
					target = scope:target_country
				}
				change_variable = { name = atd_enh_ci_done add = 1 }
			}
		}
	}
	remove_variable = atd_enh_ci_done
	if = {
		limit = { has_variable_list = atd_enh_culture_improved_this_tick }
		clear_variable_list = atd_enh_culture_improved_this_tick
	}
}

# ----- Gift-availability notify alert -------------------------------------
# Raises/clears the atd_enh_alert_giftable_leaders CMF alert when a culture
# leader is in the gift "top-up window": it owes us at least the curry cap but
# less than the gift cap, so gifts push it toward what Culture Improve spends.
# Hooked to the monthly pulse, independent of the automation toggles.
# Scope: player country (root).
atd_enh_update_gift_notify = {
	if = {
		limit = {
			has_variable = atd_enh_gift_notify_enabled
			gold >= 25
			any_country_in_diplomatic_range = {
				NOT = { this = root }
				is_discovered_by = root
				is_dominant_country_of = culture
				NOT = { is_subject_of = root }
				NOT = { is_at_war_with = root }
				prev_favors_with_this >= root.var:atd_enh_curry_favor_cap
				prev_favors_with_this < root.var:atd_enh_gift_favor_cap
			}
		}
		cmf_show_alert = { alert = atd_enh_alert_giftable_leaders }
	}
	else = {
		cmf_remove_alert = { alert = atd_enh_alert_giftable_leaders }
	}
}
