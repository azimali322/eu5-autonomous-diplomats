# ==========================================================================
# [ATD-ENH] Ask Nobility for Diplomats - optional estate automation.
# External contribution (azimali322). See atd_enh_cmm_effects.txt for the full
# list of base-file touch points and removal instructions.
#
# Replicates the vanilla ask_nobility_for_diplomats estate emergency action
# (+5 diplomats, -0.15 nobles estate satisfaction) on its 5-year cooldown,
# which is set manually via add_cooldown so the vanilla action reads it as used.
# Diplomats gate every automation in this mod, so this feeds the pool - but it
# costs estate satisfaction, so it is OFF by default and opt-in.
# Runs from the monthly hook before the favor engine, so freshly granted
# diplomats are spendable the same month.
# Scope: player country.
# ==========================================================================

atd_enh_ask_nobility_for_diplomats = {
	if = {
		limit = {
			# Settings > Enabled is the mod-wide master switch.
			has_variable = atd_enabled
			has_variable = atd_enh_ask_nobility_enabled
			country_has_estate = estate_type:nobles_estate
			NOT = { has_cooldown = ask_nobility_for_diplomats_cooldown }
		}
		add_diplomats = 5
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = -0.15
		}
		add_cooldown = {
			type = ask_nobility_for_diplomats_cooldown
			years = 5
		}
		cmf_log = { action = atd_enh_log_ask_nobility }
	}
}
