atd_enforce_on_subjects = {
	save_scope_as = actor_scope
	set_variable = { name = atd_enforce_threshold value = { value = var:atd_min_diplomats add = 1 } }
	if = {
		limit = {
			num_of_diplomats >= var:atd_enforce_threshold
		}
		cmm_for_each_list_item = {
			setting = atd__enforce
			effect = atd_enforce_each_item
		}
	}
}

atd_enforce_each_item = {
	switch = {
		trigger = scope:cmm_list_current_item_value
		flag:religion = {
			if = {
				limit = { has_variable = atd_enforce_religion_enabled }
				atd_enforce_religion_on_subjects = yes
			}
		}
		flag:culture = {
			if = {
				limit = { has_variable = atd_enforce_culture_enabled }
				atd_enforce_culture_on_subjects = yes
			}
		}
		# [ATD-ENH] Improve Subject Cultural View (item 3). Logic lives in
		# atd_enh_subject_effects.txt; remove this case to detach.
		flag:enh_improve_cultural_view = {
			if = {
				limit = {
					has_variable = atd_enh_subject_cv_enabled
					# improve_cultural_view_subject requires we lead our own culture.
					is_dominant_country_of = culture
				}
				atd_enh_improve_subject_cultural_view = yes
			}
		}
	}

	cmf_suppress = { v = $i$ }
	cmf_suppress = { v = $setting$ }
}

atd_enforce_religion_on_subjects = {
	while = {
		limit = {
			num_of_diplomats >= var:atd_enforce_threshold
			any_subject = {
				NOT = { religion = scope:actor_scope.religion }
				subject_loyalty >= 50
				liberty_desire < 50
				trigger_if = {
					limit = {
						modifier:blocked_from_conversion = yes
						has_reform = government_reform:military_order_reform
					}
					scope:actor_scope = { has_reform = government_reform:military_order_reform }
				}
				trigger_else = {
					modifier:blocked_from_conversion = no
				}
			}
		}
		count = 10
		random_subject = {
			limit = {
				NOT = { religion = scope:actor_scope.religion }
				subject_loyalty >= 50
				liberty_desire < 50
				trigger_if = {
					limit = {
						modifier:blocked_from_conversion = yes
						has_reform = government_reform:military_order_reform
					}
					scope:actor_scope = { has_reform = government_reform:military_order_reform }
				}
				trigger_else = {
					modifier:blocked_from_conversion = no
				}
			}
			save_scope_as = atd_target_subject
			enforce_religion_effect = {
				actor = scope:actor_scope
				target = scope:atd_target_subject
			}
			add_liberty_desire = liberty_desire_ultimate_plus
		}
		add_diplomats = -1
		change_variable = { name = atd_enforce_threshold add = 1 }
	}
}

atd_enforce_culture_on_subjects = {
	while = {
		limit = {
			num_of_diplomats >= var:atd_enforce_threshold
			any_subject = {
				NOT = { culture = scope:actor_scope.culture }
				subject_loyalty >= 50
				liberty_desire < 50
				# [ATD-ENH] Optionally leave culture-leader subjects for the
				# Improve Subject Cultural View action. Remove to detach.
				trigger_if = {
					limit = { scope:actor_scope = { has_variable = atd_enh_enforce_culture_skip_leaders } }
					NOT = { is_dominant_country_of = culture }
				}
			}
		}
		count = 10
		random_subject = {
			limit = {
				NOT = { culture = scope:actor_scope.culture }
				subject_loyalty >= 50
				liberty_desire < 50
				# [ATD-ENH] Same guard as the existence check above.
				trigger_if = {
					limit = { scope:actor_scope = { has_variable = atd_enh_enforce_culture_skip_leaders } }
					NOT = { is_dominant_country_of = culture }
				}
			}
			save_scope_as = atd_target_subject
			enforce_culture_effect = {
				actor = scope:actor_scope
				target = scope:atd_target_subject
			}
			add_liberty_desire = liberty_desire_ultimate_plus
		}
		add_diplomats = -1
		change_variable = { name = atd_enforce_threshold add = 1 }
	}
}
