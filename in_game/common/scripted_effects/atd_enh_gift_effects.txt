# ==========================================================================
# [ATD-ENH] Gift Reminders. See atd_enh_cmm_effects.txt for the overview.
#
# Finds the countries worth gifting right now and surfaces them in an alert
# and in the Giftable Countries window (gui/atd_enh/atd_enh_gifts_window.gui).
# The gift itself is sent by the player from the country's diplomacy screen.
# Every effect here runs in the player country's scope (root).
# ==========================================================================

# Rebuilds atd_enh_giftable_list: every country you can gift right now. The
# Giftable Countries window binds its scrolling list straight to this list.
# Rebuilt monthly and whenever the window opens, so it is current on screen.
atd_enh_rebuild_giftable_list = {
	if = {
		limit = { has_variable_list = atd_enh_giftable_list }
		clear_variable_list = atd_enh_giftable_list
	}
	every_country_in_diplomatic_range = {
		limit = { atd_enh_is_giftable_now = yes }
		root = {
			add_to_variable_list = {
				name = atd_enh_giftable_list
				target = prev
			}
		}
	}
}

# Opens or closes the Giftable Countries window (it is shown while
# atd_enh_gifts_window_open is set).
atd_enh_toggle_gifts_window = {
	if = {
		limit = { has_variable = atd_enh_gifts_window_open }
		remove_variable = atd_enh_gifts_window_open
	}
	else = {
		atd_enh_rebuild_giftable_list = yes
		set_variable = { name = atd_enh_gifts_window_open value = 1 }
	}
}

# Refreshes the list, then raises the alert variant matching how many
# countries are giftable, so its tooltip can name up to three of them. Only
# raises an alert while Settings > Enabled (the mod-wide master switch) and
# Notify When Giftable are both on, and the treasury is not nearly empty.
atd_enh_update_gift_notify = {
	atd_enh_rebuild_giftable_list = yes

	remove_variable = atd_enh_gift_t1
	remove_variable = atd_enh_gift_t2
	remove_variable = atd_enh_gift_t3
	set_variable = { name = atd_enh_gift_n value = 0 }

	if = {
		limit = {
			has_variable = atd_enabled
			has_variable = atd_enh_gift_notify_enabled
			gold >= 25
			has_variable_list = atd_enh_giftable_list
		}
		every_in_list = {
			variable = atd_enh_giftable_list
			root = {
				change_variable = { name = atd_enh_gift_n add = 1 }
				if = {
					limit = { NOT = { has_variable = atd_enh_gift_t1 } }
					set_variable = { name = atd_enh_gift_t1 value = prev }
				}
				else_if = {
					limit = { NOT = { has_variable = atd_enh_gift_t2 } }
					set_variable = { name = atd_enh_gift_t2 value = prev }
				}
				else_if = {
					limit = { NOT = { has_variable = atd_enh_gift_t3 } }
					set_variable = { name = atd_enh_gift_t3 value = prev }
				}
			}
		}
	}

	# Show only the variant that matches the count, and clear the others so a
	# stale one cannot sit on the bar naming countries that are no longer
	# giftable.
	if = {
		limit = { var:atd_enh_gift_n >= 4 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
		cmf_show_alert = { alert = atd_enh_alert_giftable_more }
	}
	else_if = {
		limit = { var:atd_enh_gift_n = 3 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
		cmf_show_alert = { alert = atd_enh_alert_giftable_3 }
	}
	else_if = {
		limit = { var:atd_enh_gift_n = 2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
		cmf_show_alert = { alert = atd_enh_alert_giftable_2 }
	}
	else_if = {
		limit = { var:atd_enh_gift_n = 1 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
		cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
		cmf_show_alert = { alert = atd_enh_alert_giftable_1 }
	}
	else = {
		atd_enh_clear_gift_alerts = yes
	}
}

# Removes every gift alert.
atd_enh_clear_gift_alerts = {
	cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
	cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
	cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
	cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
}
