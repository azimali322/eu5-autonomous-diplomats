# Callback handler for cmf_on_callback.
# var:cmf_callback is the flag of the setting, alert, or action bar element that was interacted with.
# Scope: country
atd_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# Improve Relations (improve_relations) Tab
		## Automation (automation) Group
		flag:atd__improve_relations_enabled = {
			cmm_sync_bool_alias = {
				setting = atd__improve_relations_enabled
				alias = atd_improve_relations_enabled
			}
		}
		## Limits (limits) Group
		flag:atd__max_relations = {
			cmm_sync_setting_alias = {
				setting = atd__max_relations
				alias = atd_max_relations
			}
		}
		flag:atd__max_antagonism = {
			cmm_sync_setting_alias = {
				setting = atd__max_antagonism
				alias = atd_max_antagonism
			}
		}
		# Settings (settings) Tab
		## Automation (automation) Group
		flag:atd__enabled = {
			cmm_sync_bool_alias = {
				setting = atd__enabled
				alias = atd_enabled
			}
			if = {
				limit = { has_variable = atd_enabled }
				set_automated_system = {
					system = diplomacy
				}
			}
			else = {
				set_automated_system = {
					system = diplomacy
					activate = no
				}
			}
		}
		## Reserve (reserve) Group
		flag:atd__min_diplomats = {
			# The CMM slider fires this callback on hover as well as on change.
			set_local_variable = { name = atd_tmp_flag value = flag:atd__min_diplomats }
			if = {
				limit = {
					NOT = { "variable_map(cmm|local_var:atd_tmp_flag)" = var:atd_min_diplomats }
				}
				cmm_sync_setting_alias = {
					setting = atd__min_diplomats
					alias = atd_min_diplomats
				}
				# Picked up by the push widgets in atd_automation_lateralview.gui.
				set_variable = atd_reserve_from_cmm
			}
		}
		flag:atd__min_diplomat_gain = {
			cmm_sync_setting_alias = {
				setting = atd__min_diplomat_gain
				alias = atd_min_diplomat_gain
			}
		}
	}

	# Benign warning suppression
	if = {
		limit = { always = no }
		cmf_suppress = { v = atd_tmp_flag }
		cmf_suppress = { v = atd_tmp_reserve }
	}
}

# Aligns the CMM reserve slider's range with the observed engine slider scale.
# Scope: country
atd_update_reserve_slider_bounds = {
	if = {
		limit = { has_variable = atd_reserve_slider_max }
		remove_from_variable_map = { name = cmm_max key = flag:atd__min_diplomats }
		add_to_variable_map = { name = cmm_max key = flag:atd__min_diplomats value = var:atd_reserve_slider_max }

		set_variable = { name = atd_tmp_step_count value = var:atd_reserve_slider_max }
		change_variable = { name = atd_tmp_step_count divide = 0.1 }
		round_variable = {
			name = atd_tmp_step_count
			nearest = 1
		}
		change_variable = { name = atd_tmp_step_count add = 1 }
		if = {
			limit = { var:atd_tmp_step_count < 2 }
			set_variable = { name = atd_tmp_step_count value = 2 }
		}
		remove_from_variable_map = { name = cmm_slider_actual_step_count key = flag:atd__min_diplomats }
		add_to_variable_map = { name = cmm_slider_actual_step_count key = flag:atd__min_diplomats value = var:atd_tmp_step_count }

		change_variable = { name = atd_tmp_step_count add = -1 }
		remove_from_variable_map = { name = cmm_slider_actual_last_index key = flag:atd__min_diplomats }
		add_to_variable_map = { name = cmm_slider_actual_last_index key = flag:atd__min_diplomats value = var:atd_tmp_step_count }
		remove_variable = atd_tmp_step_count
	}
}

# Sets a CMM bool setting and fires the shared callback.
# Scope: country
# Args: setting (key), value (1/0)
atd_set_bool_setting = {
	remove_from_variable_map = { name = cmm key = flag:$setting$ }
	add_to_variable_map = { name = cmm key = flag:$setting$ value = $value$ }
	set_variable = { name = cmf_callback value = flag:$setting$ }
	trigger_event_silently = {
		on_action = cmf_on_callback
	}
	remove_variable = cmf_callback
}
