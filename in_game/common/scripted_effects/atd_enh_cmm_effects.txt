# ==========================================================================
# [ATD-ENH] Favor-economy / subject cultural-view enhancements - CMM wiring.
# External contribution (azimali322). All enhancement logic lives in atd_enh_*
# files; the only touch points inside the base mod are marked with [ATD-ENH]
# (grep for that tag). To remove the whole feature set: delete the atd_enh_*
# files and revert those marked lines.
#
# Touch points in base files:
#   atd_cmm_effects.txt     - enforce item_count 2 -> 3,
#                             one call to atd_enh_register
#   atd_enforce_effects.txt - 1 switch case (enh_improve_cultural_view),
#                             2 skip-leaders guards in enforce culture
#
# The Favors tab is entirely enhancement-owned: an Automation group (master
# toggle + Curry Favors + Culture Improve), a Limits group (favor caps), a
# priority-ordered target-category list mirroring the Improve Relations
# categories, and a Gifts group that drives the gift REMINDER only.
# NOTE: gifts cannot be automated. perform_diplomatic_action rejects send_gift
# ("Cannot find send_gift in country_interactions database or list of hardcoded
# diplo actions", country_effects.cpp:4422), because it is built around its own
# amount-slider window. The mod therefore names giftable countries in an alert
# and the player sends the gift manually, choosing the amount.
# ==========================================================================

# Callback handler for the enhancement settings. Hooked to cmf_on_callback via
# atd_enh_on_actions.txt (runs alongside the base mod's handler; flags each
# handler doesn't know are ignored by its switch).
# Scope: country
atd_enh_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# Action bar: open/close the Giftable Countries window.
		flag:atd_enh_open_gifts = {
			atd_enh_toggle_gifts_window = yes
		}
		# Alert click opens the window. CMF clears an alert on left-click, so
		# re-raise it afterwards - it is a standing reminder, not a one-shot.
		flag:atd_enh_alert_giftable_1 = {
			atd_enh_toggle_gifts_window = yes
			atd_enh_update_gift_notify = yes
		}
		flag:atd_enh_alert_giftable_2 = {
			atd_enh_toggle_gifts_window = yes
			atd_enh_update_gift_notify = yes
		}
		flag:atd_enh_alert_giftable_3 = {
			atd_enh_toggle_gifts_window = yes
			atd_enh_update_gift_notify = yes
		}
		flag:atd_enh_alert_giftable_more = {
			atd_enh_toggle_gifts_window = yes
			atd_enh_update_gift_notify = yes
		}
		# Favors tab - Automation
		flag:atd__enh_favors_enabled = {
			cmm_sync_bool_alias = {
				setting = atd__enh_favors_enabled
				alias = atd_enh_favors_enabled
			}
		}
		flag:atd__enh_curry_enabled = {
			cmm_sync_bool_alias = {
				setting = atd__enh_curry_enabled
				alias = atd_enh_curry_enabled
			}
		}
		flag:atd__enh_culture_improve_enabled = {
			cmm_sync_bool_alias = {
				setting = atd__enh_culture_improve_enabled
				alias = atd_enh_culture_improve_enabled
			}
		}
		flag:atd__enh_culture_improve_count = {
			cmm_sync_setting_alias = {
				setting = atd__enh_culture_improve_count
				alias = atd_enh_culture_improve_count
			}
		}
		# Favors tab - Limits
		flag:atd__enh_curry_favor_cap = {
			cmm_sync_setting_alias = {
				setting = atd__enh_curry_favor_cap
				alias = atd_enh_curry_favor_cap
			}
		}
		flag:atd__enh_gift_favor_cap = {
			cmm_sync_setting_alias = {
				setting = atd__enh_gift_favor_cap
				alias = atd_enh_gift_favor_cap
			}
		}
		# Favors tab - Gifts
		flag:atd__enh_gift_notify_enabled = {
			cmm_sync_bool_alias = {
				setting = atd__enh_gift_notify_enabled
				alias = atd_enh_gift_notify_enabled
			}
			# Reflect the toggle on the alert straight away.
			atd_enh_update_gift_notify = yes
		}
		# Settings tab - Estate
		flag:atd__enh_ask_nobility_enabled = {
			cmm_sync_bool_alias = {
				setting = atd__enh_ask_nobility_enabled
				alias = atd_enh_ask_nobility_enabled
			}
		}
		# Subjects tab - Options
		flag:atd__enh_enforce_culture_skip_leaders = {
			cmm_sync_bool_alias = {
				setting = atd__enh_enforce_culture_skip_leaders
				alias = atd_enh_enforce_culture_skip_leaders
			}
		}
	}
}
