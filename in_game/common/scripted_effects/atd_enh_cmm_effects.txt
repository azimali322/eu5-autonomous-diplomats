# ==========================================================================
# [ATD-ENH] Autonomous Diplomats Enhanced - settings and callbacks.
# Enhancement layer by azimali322 on top of Conner Olsen's Autonomous Diplomats.
# Everything lives in the atd_enh_* files. The only edits inside the base mod
# are tagged [ATD-ENH] (grep for the tag); delete the atd_enh_* files and revert
# those lines to remove the layer completely:
#   atd_cmm_effects.txt     - one call to atd_enh_register (end of
#                             atd_register_cmf_mod), and the Subject Actions
#                             list's item_count raised from 2 to 3
#   atd_enforce_effects.txt - one switch case (enh_improve_cultural_view) and
#                             two Skip Subject Culture Leaders guards in
#                             atd_enforce_culture_on_subjects
#
# Cultural View Reminders (Auto Interactions tab): an alert, an action-bar
# button and a clickable window listing the countries whose culture's view of
# yours you can improve right now with Improve Cultural View - you have the 50
# favors it costs. The action itself stays manual. Pairs with the base mod's
# Auto-Gifting, which builds the favors.
#
# Improve Cultural View (Subjects tab, row 3 of Subject Actions): replicates
# the vanilla improve_cultural_view_subject action (atd_enh_subject_effects.txt).
# Enforce Culture: Skip Subject Culture Leaders (Subjects tab, Options): leaves
# subjects that lead their own culture to Improve Cultural View instead.
# ==========================================================================

# Registers the enhancement settings. Called from the end of
# atd_register_cmf_mod. Root scope: country.
atd_enh_register = {
	# Action-bar button that opens the Cultural Views window (idempotent).
	cmf_add_action_bar_element = { element = atd_enh_open_culture_view }

	# ----- Auto Interactions (interactions) Tab -----------------------------
	## Cultural View Reminders (enh_culture_view) Group
	cmm_register_bool_setting = {
		mod_id = atd
		setting_id = enh_culture_view_notify_enabled
		tab_id = interactions
		group_id = enh_culture_view
		default_value = 1
	}
	cmm_sync_bool_alias = {
		setting = atd__enh_culture_view_notify_enabled
		alias = atd_enh_culture_view_notify_enabled
	}

	# ----- Subjects (subjects) Tab ------------------------------------------
	# Subject Actions list, row 3 (the base list's item_count is raised to 3).
	# Its checkbox is re-read each month in atd_enh_improve_subject_cultural_view.
	cmm_set_list_item_value = {
		mod_id = atd
		setting_id = enforce
		item = 3
		value = flag:enh_improve_cultural_view
	}
	cmm_sync_bool_alias = {
		setting = atd__enforce_i3_f1
		alias = atd_enh_subject_cv_enabled
	}

	## Options (enh_options) Group
	cmm_register_bool_setting = {
		mod_id = atd
		setting_id = enh_enforce_culture_skip_leaders
		tab_id = subjects
		group_id = enh_options
		default_value = 0
	}
	cmm_sync_bool_alias = {
		setting = atd__enh_enforce_culture_skip_leaders
		alias = atd_enh_enforce_culture_skip_leaders
	}
}

# Callback handler for our settings, alerts and action-bar button. Hooked to
# cmf_on_callback in atd_enh_on_actions.txt; it runs alongside the base mod's
# handler, and each handler ignores flags it does not know.
# Scope: country
atd_enh_handle_cmf_callback = {
	switch = {
		trigger = var:cmf_callback
		# Action-bar button: open or close the Cultural Views window.
		flag:atd_enh_open_culture_view = {
			atd_enh_toggle_culture_view_window = yes
		}
		# Left-clicking an alert opens the window. CMF clears an alert on
		# left-click, so raise it again - it is a standing reminder.
		flag:atd_enh_alert_culture_view_1 = {
			atd_enh_toggle_culture_view_window = yes
			atd_enh_update_culture_view_notify = yes
		}
		flag:atd_enh_alert_culture_view_2 = {
			atd_enh_toggle_culture_view_window = yes
			atd_enh_update_culture_view_notify = yes
		}
		flag:atd_enh_alert_culture_view_3 = {
			atd_enh_toggle_culture_view_window = yes
			atd_enh_update_culture_view_notify = yes
		}
		flag:atd_enh_alert_culture_view_more = {
			atd_enh_toggle_culture_view_window = yes
			atd_enh_update_culture_view_notify = yes
		}
		## Auto Interactions tab - Cultural View Reminders group
		flag:atd__enh_culture_view_notify_enabled = {
			cmm_sync_bool_alias = {
				setting = atd__enh_culture_view_notify_enabled
				alias = atd_enh_culture_view_notify_enabled
			}
			# Show or clear the alert straight away.
			atd_enh_update_culture_view_notify = yes
		}
		## Subjects tab - Options group
		flag:atd__enh_enforce_culture_skip_leaders = {
			cmm_sync_bool_alias = {
				setting = atd__enh_enforce_culture_skip_leaders
				alias = atd_enh_enforce_culture_skip_leaders
			}
		}
	}
}
