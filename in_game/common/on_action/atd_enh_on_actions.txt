# ==========================================================================
# [ATD-ENH] On-action hooks for the enhancement features.
# External contribution (azimali322). See atd_enh_cmm_effects.txt for the full
# list of base-file touch points and removal instructions.
#
# These hooks run alongside the base mod's own hooks on the same CMF
# on_actions; deleting this file detaches the favor engine, callback handling,
# and the gift-availability alert without touching base files.
# ==========================================================================

# Handle setting changes for the enhancement settings.
cmf_on_callback = {
	on_actions = {
		atd_enh_on_cmf_callback
	}
}

atd_enh_on_cmf_callback = {
	effect = {
		atd_enh_handle_cmf_callback = yes
	}
}

# Monthly: run the favor engine, then refresh the gift-availability alert
# (the alert is independent of the automation toggles, so it also serves
# manual play).
cmf_monthly_human_country_pulse = {
	on_actions = {
		atd_enh_on_monthly
	}
}

atd_enh_on_monthly = {
	effect = {
		atd_enh_favor_economy = yes
		atd_enh_update_gift_notify = yes
	}
}
