# ==========================================================================
# [ATD-ENH] On-action hooks for the enhancement features.
# External contribution (azimali322). See atd_enh_cmm_effects.txt for the full
# list of base-file touch points and removal instructions.
#
# These hooks run alongside the base mod's own hooks on the same CMF
# on_actions; deleting this file detaches the favor engine, callback handling,
# and the gift-availability alert without touching base files.
# ==========================================================================

# Handle setting changes for the enhancement settings.
cmf_on_callback = {
	on_actions = {
		atd_enh_on_cmf_callback
	}
}

atd_enh_on_cmf_callback = {
	effect = {
		atd_enh_handle_cmf_callback = yes
	}
}

# Monthly: refresh the gift-availability alert, and run the favor engine
# standalone ONLY when the base mod's pass will not run this month (its
# Settings > Enabled toggle is off, so the hidden-window driver - which
# normally runs the prioritized pass including favors - never opens). This
# keeps favors usable with the base automation disabled without double-running.
cmf_monthly_human_country_pulse = {
	on_actions = {
		atd_enh_on_monthly
	}
}

# Everything here is gated on Settings > Enabled (atd_enabled), the mod-wide
# master switch. The favor engine itself is NOT run here: it runs inside the
# base mod's monthly pass via atd_enh_run_prioritized_pass, so the Automation
# Priority order applies and it cannot run twice.
atd_enh_on_monthly = {
	effect = {
		if = {
			limit = { has_variable = atd_enabled }
			# Runs before the base pass so freshly granted diplomats are
			# spendable the same month.
			atd_enh_ask_nobility_for_diplomats = yes
			atd_enh_update_gift_notify = yes
		}
		else = {
			# Master off: clear anything we might have raised.
			cmf_remove_alert = { alert = atd_enh_alert_giftable_1 }
			cmf_remove_alert = { alert = atd_enh_alert_giftable_2 }
			cmf_remove_alert = { alert = atd_enh_alert_giftable_3 }
			cmf_remove_alert = { alert = atd_enh_alert_giftable_more }
		}
	}
}
