# ==========================================================================
# [ATD-ENH] On-action hooks for the enhancement layer.
# See atd_enh_cmm_effects.txt for the overview.
#
# These run alongside the base mod's hooks on the same CMF on_actions (on_action
# lists merge), so deleting this file detaches the layer without touching base
# files.
# ==========================================================================

# Setting changes, alert clicks and the action-bar button.
cmf_on_callback = {
	on_actions = {
		atd_enh_on_cmf_callback
	}
}

atd_enh_on_cmf_callback = {
	effect = {
		atd_enh_handle_cmf_callback = yes
	}
}

# Monthly. The Cultural Views work itself (auto improve, curry, reminder
# refresh) runs inside the base mod's monthly pass, which only runs while
# Settings > Enabled - the mod-wide master switch - is on. Here:
# - Enabled on: clear atd_enh_currying_seen before the base mod opens its
#   hidden-window cycle, so the curry trackers refill it with this month's
#   curries (as the base mod's atd_gui_sync_begin does for its own list), and
#   rebuild atd_enh_curry_watch, the countries atd_enh_curry_tracker.gui checks.
# - Enabled on: also schedule a Cultural Views check for each of the next 30
#   days (atd_enh_daily_check), so Auto Improve and the reminder act the day a
#   country reaches 50 favors - the 50 can decay away within the month.
# - Enabled off: clear the alerts and close the window.
cmf_monthly_human_country_pulse = {
	on_actions = {
		atd_enh_on_monthly
	}
}

atd_enh_on_monthly = {
	effect = {
		if = {
			limit = { has_variable = atd_enabled }
			if = {
				limit = { has_variable_list = atd_enh_currying_seen }
				clear_variable_list = atd_enh_currying_seen
			}
			if = {
				limit = { has_variable_list = atd_enh_curry_watch }
				clear_variable_list = atd_enh_curry_watch
			}
			if = {
				limit = { has_variable = atd_enh_curry_cv_enabled }
				every_country_in_diplomatic_range = {
					limit = { is_dominant_country_of = culture }
					root = {
						add_to_variable_list = {
							name = atd_enh_curry_watch
							target = prev
						}
					}
				}
			}
			atd_enh_schedule_daily_checks = yes
		}
		else = {
			atd_enh_clear_culture_view_alerts = yes
			cmf_remove_alert = { alert = atd_enh_alert_culture_view_improved }
			if = {
				limit = { has_variable = atd_enh_culture_view_window_open }
				remove_variable = atd_enh_culture_view_window_open
			}
		}
	}
}

# Daily Cultural Views check, scheduled by atd_enh_schedule_daily_checks.
# Auto Improve Cultural View plus the reminder; gated on Settings > Enabled
# inside atd_enh_cultural_view_daily.
atd_enh_daily_check = {
	effect = {
		atd_enh_cultural_view_daily = yes
	}
}
