# ==========================================================================
# [ATD-ENH] Target-eligibility triggers for the enhancement actions.
# External contribution (azimali322). See atd_enh_cmm_effects.txt for the full
# list of base-file touch points and removal instructions.
#
# Evaluated in the CANDIDATE country's scope; scope:actor_scope is the player
# (saved by atd_enh_favor_economy / atd_enforce_on_subjects). All respect the
# base mod's Block Autonomous Improving flag (atd_improve_blocked). In picker
# limits, prev = the player, so prev_favors_with_this = favors the candidate
# owes the player.
# ==========================================================================

# A country we can build favors with this tick using at least one enabled
# method. Category membership (neighbor, ally, ...) is checked by the caller.
atd_enh_is_favor_candidate = {
	NOT = { this = scope:actor_scope }
	is_discovered_by = scope:actor_scope
	NOT = { is_at_war_with = scope:actor_scope }
	NOT = { has_variable = atd_improve_blocked }
	scope:actor_scope = {
		OR = {
			NOT = { has_variable_list = atd_enh_favor_done_this_tick }
			NOT = { is_target_in_variable_list = { name = atd_enh_favor_done_this_tick target = prev } }
		}
	}
	OR = {
		AND = {
			scope:actor_scope = { has_variable = atd_enh_curry_enabled }
			atd_enh_can_curry_target = yes
		}
		AND = {
			scope:actor_scope = { has_variable = atd_enh_gifts_enabled }
			atd_enh_can_gift_target = yes
		}
	}
}

# Curry (influence_nation) eligibility: non-great-power owing fewer favors than
# the curry cap, and the actor is allowed to use Influence Nation.
atd_enh_can_curry_target = {
	is_great_power = no
	prev_favors_with_this < scope:actor_scope.var:atd_enh_curry_favor_cap
	scope:actor_scope = {
		modifier:allow_diplomacy_influence_nation = yes
	}
}

# Gift (send_gift) eligibility: owes fewer favors than the gift cap, enough gold
# to cover the minimum gift, and (optionally) a culture leader only. There is no
# spend budget here by design - the game's own Send Gift window asks the player
# for the amount (or lets them cancel), so the player sets the spend per gift.
# The per-category Actions sliders cap how many gifts are offered each month.
#
# Send Gift is hardcoded with a PER-TARGET cooldown of
# define:NDiplomacy|SEND_GIFT_COOLDOWN (120 months / 10 years). It has no
# cooldown = {} block and no has_cooldown token - the only way to read it is the
# pairwise trigger has_gifted_gold_to, checked from the SENDER's scope. Without
# this the engine burns its monthly attempts re-gifting a target the game is
# still blocking. (Mechanic confirmed against the Cooldown Notifier mod, which
# watches this same cooldown.)
atd_enh_can_gift_target = {
	prev_favors_with_this < scope:actor_scope.var:atd_enh_gift_favor_cap
	trigger_if = {
		limit = { scope:actor_scope = { has_variable = atd_enh_gift_culture_leaders_only } }
		is_dominant_country_of = culture
	}
	scope:actor_scope = {
		# The minimum possible gift (prices/00_hardcoded.txt min_scale = 25).
		gold >= 25
		# 10-year per-target gift cooldown still running.
		NOT = { has_gifted_gold_to = prev }
	}
}

# A country the player could send a gift to right now, for the notify alert.
# Evaluated in the candidate's scope from root (the player), so prev = player.
# Deliberately wider than the automation's own gift check: any dominant country
# of a primary culture that is off its gift cooldown and still owes less than
# the gift cap, since gifts can build favors from zero, not just top up.
atd_enh_is_giftable_now = {
	NOT = { this = root }
	is_discovered_by = root
	is_dominant_country_of = culture
	NOT = { is_subject_of = root }
	NOT = { is_at_war_with = root }
	NOT = { has_variable = atd_improve_blocked }
	prev_favors_with_this < root.var:atd_enh_gift_favor_cap
	root = {
		NOT = { has_gifted_gold_to = prev }
	}
}

# An improve_cultural_view target: a culture leader whose culture does not
# already view ours as kindred, not improved this tick.
atd_enh_is_culture_target = {
	NOT = { this = scope:actor_scope }
	is_discovered_by = scope:actor_scope
	is_dominant_country_of = culture
	NOT = { is_subject_of = scope:actor_scope }
	NOT = { is_at_war_with = scope:actor_scope }
	NOT = { has_variable = atd_improve_blocked }
	NOT = {
		culture = {
			cultural_view = {
				target = scope:actor_scope.culture
				value = kindred
			}
		}
	}
	scope:actor_scope = {
		OR = {
			NOT = { has_variable_list = atd_enh_culture_improved_this_tick }
			NOT = { is_target_in_variable_list = { name = atd_enh_culture_improved_this_tick target = prev } }
		}
	}
}

# An Improve Subject Cultural View target. Evaluated in the SUBJECT's scope.
# Mirrors the vanilla improve_cultural_view_subject select_trigger (dominant of
# its culture, not already kindred toward us, loyal, low liberty desire) plus an
# off-cooldown check (the action has a 50-year per-subject cooldown).
atd_enh_is_subject_cv_target = {
	is_dominant_country_of = culture
	NOT = {
		culture = {
			cultural_view = {
				target = scope:actor_scope.culture
				value = kindred
			}
		}
	}
	subject_loyalty >= 50
	liberty_desire < 50
	NOT = { has_cooldown = improve_cultural_view_subject }
}
