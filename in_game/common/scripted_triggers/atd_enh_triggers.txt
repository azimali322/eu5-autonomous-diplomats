# ----- Cultural Views -------------------------------------------------------
# All of these are evaluated in a candidate country's scope with the player as
# root (and as prev: callers reach the candidate from the player through an
# in-diplomatic-range iterator or the player's own lists). Callers only run
# them while the player leads its own culture, which Improve Cultural View
# requires (its potential block).
#
# prev_favors_with_this is PREV's favors with THIS - the game's trigger text
# reads "[TARGET]'s favors with [COUNTRY]" - so here it is the player's favors
# with the candidate: what Curry Favors builds and Improve Cultural View spends.
#
# The fixed numbers mirror 1.4.1 game rules; local/tests/validate_mod.py check
# 11 re-reads the game files and fails if a patch changes any of them:
#   50 favors - improve_cultural_view_cost (diplomatic_costs/01_from_script.txt)
#   opinion   - improve_cultural_view's diplo_chance: base -10, +0.2 per point
#               of their opinion of us, so they only accept above 50
#   rivals    - its accept block: -100 when they count us as a rival
#   trust 50  - favors can only be curried up to their trust in us
#               (CURRY_FAVORS_LOW_TRUST), and the curry cap
#               CURRY_FAVORS_MAX_FAVORS is 50

# A country whose culture's view of yours Improve Cultural View can raise, and
# who would accept it (improve_cultural_view.txt):
#   visible - it leads its culture and is not our subject (subjects have their
#             own action: Subject Actions row 3)
#   enabled - its culture does not already see ours as kindred
#   accept  - their opinion of us is above 50 and neither side counts the other
#             as a rival (the game checks their side; we skip ours too)
atd_enh_cultural_view_target = {
	NOT = { this = root }
	is_discovered_by = root
	is_dominant_country_of = culture
	NOT = { is_subject_of = root }
	NOT = { is_at_war_with = root }
	NOT = { is_rival_of = root }
	root = { NOT = { is_rival_of = prev } }
	NOT = {
		culture = {
			cultural_view = {
				target = root.culture
				value = kindred
			}
		}
	}
	opinion = { target = root value > 50 }
}

# A country you can use Improve Cultural View on right now: you also have the
# 50 favors it costs. Used by the reminder and by Auto Improve Cultural View.
atd_enh_can_improve_cultural_view = {
	atd_enh_cultural_view_target = yes
	prev_favors_with_this >= 50
}

# After an attempt: its culture still views ours at the level recorded just
# before it (atd_enh_save_cv_view), so the attempt did not go through.
# Scope: the target country. Root: the player.
atd_enh_cv_view_unchanged = {
	OR = {
		AND = {
			var:atd_enh_cv_view_before = flag:enemy
			culture = { cultural_view = { target = root.culture value = enemy } }
		}
		AND = {
			var:atd_enh_cv_view_before = flag:negative
			culture = { cultural_view = { target = root.culture value = negative } }
		}
		AND = {
			var:atd_enh_cv_view_before = flag:neutral
			culture = { cultural_view = { target = root.culture value = neutral } }
		}
		AND = {
			var:atd_enh_cv_view_before = flag:positive
			culture = { cultural_view = { target = root.culture value = positive } }
		}
		AND = {
			var:atd_enh_cv_view_before = flag:kindred
			culture = { cultural_view = { target = root.culture value = kindred } }
		}
	}
}

# Auto Improve Cultural View's pick: as above, and not tried in the last two
# months (atd_enh_cv_attempt, set by every attempt). Favors drop by 50 when it
# works, but one that does not go through leaves them at 50 - and before this
# wait, the daily check repeated such an attempt every day, spending a diplomat
# each time (reported 2026-10-18: 8 tries on Kanem, 6 on Mali).
atd_enh_can_auto_improve_now = {
	atd_enh_can_improve_cultural_view = yes
	NOT = { has_variable = atd_enh_cv_attempt }
}

# A country Curry Favors for Cultural Views should start currying now. On top
# of being a cultural view target, the game's own curry rules
# (CURRY_FAVORS_* in loc/defines): not anyone's subject, their opinion of us at
# least CURRY_FAVORS_RELATION_REQUIRED (10, covered by the 50 above), and
# favors below both the curry cap and their trust in us - so trust must be at
# least 50 for the curry to reach the 50 the improvement costs.
atd_enh_can_start_curry = {
	atd_enh_cultural_view_target = yes
	is_subject = no
	trust = { target = root value >= 50 }
	# With Auto Improve on, a country is curried whenever it is below 50, since
	# the 50 is spent the month it is reached. With it off, a country that
	# reached 50 is left alone until its favors decay below 45, so diplomats are
	# not spent topping it back up while the player decides.
	trigger_if = {
		limit = { root = { has_variable = atd_enh_auto_improve_cv_enabled } }
		prev_favors_with_this < 50
	}
	trigger_else = {
		prev_favors_with_this < 45
	}
	# Not already being curried: reported by a curry tracker this month, or
	# started earlier in this pass.
	root = {
		OR = {
			NOT = { has_variable_list = atd_enh_currying_seen }
			NOT = {
				is_target_in_variable_list = {
					name = atd_enh_currying_seen
					target = prev
				}
			}
		}
	}
	# Safety net: not started in the last 6 months (cleared early when Auto
	# Improve spends the favors). If the trackers ever failed, nothing would be
	# "seen" and the mod would otherwise retry the same country every month.
	NOT = { has_variable = atd_enh_curry_attempt }
}

# The candidate's culture makes up at least 1% of your own population, so a
# better view of your culture matters at home. Curry Favors tries these first.
atd_enh_culture_is_in_our_country = {
	root = {
		culture_percentage_in_country = {
			culture = prev.culture
			value >= 0.01
		}
	}
}

# A subject Improve Cultural View can target. Mirrors the select_trigger of the
# vanilla improve_cultural_view_subject interaction, using the same named
# values for the loyalty and liberty gates so a game update that changes them
# carries over. Scope: the subject. Reads scope:atd_actor_scope (us).
atd_enh_is_subject_cv_target = {
	is_dominant_country_of = culture
	NOT = {
		culture = {
			cultural_view = {
				target = scope:atd_actor_scope.culture
				value = kindred
			}
		}
	}
	subject_loyalty >= MIN_SUBJECT_LOYALTY_FOR_ACTION
	liberty_desire < MAX_SUBJECT_LIBERTY_DESIRE_FOR_ACTION
}

# Skip Subject Culture Leaders: while the option is on, Enforce Culture leaves
# subjects that are the dominant country of their culture alone, so Improve
# Cultural View can work on them instead. Used by the two tagged guards in the
# base atd_enforce_culture_on_subjects. Scope: the subject.
atd_enh_enforce_culture_allowed = {
	trigger_if = {
		limit = { scope:atd_actor_scope = { has_variable = atd_enh_enforce_culture_skip_leaders } }
		NOT = { is_dominant_country_of = culture }
	}
}
