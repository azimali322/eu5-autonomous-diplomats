# Clears the atd_pending_cancel variable on the country in scope. Called from
# the auto-cancel widget's state in outliner_entries.gui right after firing
# OutlinerDiplomacyEntry.OnRightClick, so the same widget doesn't keep firing
# the cancel during the window before sync removes the entry from tracking.
# Root: target country (set via SetRoot).
atd_clear_pending_cancel = {
	effect = {
		if = {
			limit = { has_variable = atd_pending_cancel }
			remove_variable = atd_pending_cancel
		}
		# Set the auto-confirm flag as a GLOBAL variable. The popup that
		# OutlinerDiplomacyEntry.OnRightClick puts up is confirm_diplomatic_action_popup
		# (in_game/gui/ingame_dialogs.gui), whose DialogConfig context doesn't
		# expose the target country - so we can't gate per-country anyway.
		# A single global flag is sufficient: at most one cancel is pending at
		# a time, and the popup's auto-confirm widget clears the flag the
		# moment it fires DialogConfig.OnAccept.
		# Global also dodges the "what scope are we in" issue here - root is
		# the target country (from SetRoot in the outliner widget call), and
		# scope-switching to player from script effects is awkward.
		if = {
			limit = { NOT = { has_global_variable = atd_pending_confirm } }
			set_global_variable = atd_pending_confirm
		}
	}
}

# Cleared by the confirm_diplomatic_action_popup auto-confirm widget right
# after firing DialogConfig.OnAccept, so the same widget doesn't keep firing
# on subsequent polls while a stale flag would otherwise still match.
# Scope: any (operates on global variables).
atd_clear_pending_confirm = {
	effect = {
		if = {
			limit = { has_global_variable = atd_pending_confirm }
			remove_global_variable = atd_pending_confirm
		}
	}
}

# Runs from the hidden window driver after the iter trigger has populated
# atd_improving_seen. Reconciles, runs mod logic. Does NOT clear
# atd_pending_action - that happens in atd_close_hidden_window after the
# auto-cancel state has had its turn.
# Root: player country.
atd_gui_sync_step = {
	effect = {
		atd_gui_sync_end = yes
		# [ATD-ENH] Run Subjects / Favors / Improve Relations in the user's
		# Automation Priority order (Settings tab) instead of the fixed order.
		# To detach: replace this call with the original two lines:
		#   atd_enforce_on_subjects = yes
		#   if = { limit = { has_variable = atd_improve_relations_enabled } atd_improve_relations = yes }
		atd_enh_run_prioritized_pass = yes
	}
}

# Final state of the driver chain. Drops atd_pending_action so the gate
# closes and the widget tears down until the next monthly tick.
# Root: player country.
atd_close_hidden_window = {
	effect = {
		remove_variable = atd_pending_action
	}
}

# Visibility gate for atd_hidden_window. Set by atd_on_monthly, cleared by
# atd_close_hidden_window. Root: player country.
atd_should_run = {
	is_shown = {
		has_variable = atd_pending_action
	}
}

# Per-country tracking call. Fired once per ongoing improve_opinion country
# during the begin phase via PdxGuiTriggerAllAnimations. Adds the country to
# the seen list so the end phase knows this entry is still active.
#
# We don't pull GUI progress into atd_improved_with anymore: passing
# OngoingRelationCountry.GetOpinionImprovementProgress through MakeScopeValue
# in the on_finish chain made the GUI engine error with "Promote 'AddScope'
# returned nullptr". The skip_threshold cooldown that depended on progress is
# now redundant anyway - an opinion-cap cancel sets a 4-year
# atd_cancel_cooldown, which covers the only case the cooldown was needed
# for (the auto-cancel retarget loop).
# Root: player country. Saved scopes: country (target).
atd_gui_sync_track = {
	saved_scopes = { country }
	effect = {
		if = {
			limit = {
				OR = {
					NOT = { has_variable_list = atd_improving_seen }
					NOT = {
						is_target_in_variable_list = {
							name = atd_improving_seen
							target = scope:country
						}
					}
				}
			}
			add_to_variable_list = {
				name = atd_improving_seen
				target = scope:country
			}
		}
	}
}
