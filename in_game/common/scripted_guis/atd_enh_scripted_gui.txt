# Close button on the Giftable Countries window.
atd_enh_close_gifts_window = {
	scope = country

	effect = {
		if = {
			limit = { has_variable = atd_enh_gifts_window_open }
			remove_variable = atd_enh_gifts_window_open
		}
	}
}
